Apps Script para conectar con Google Sheets
Pegá acá la URL que te da Google Apps Script al publicar como Web App

Cómo crear el Apps Script

  1. Abrí tu Google Sheet
  2. Hacé clic en Extensiones → Apps Script
  3. Borrá todo el código que aparece y pegá el código de abajo
  4. Hacé clic en Implementar → Nueva implementación
  5. Tipo: Aplicación web
  6. Ejecutar como: Yo
  7. Quién tiene acceso: Cualquier persona
  8. Hacé clic en Implementar y copiá la URL que aparece
  9. Pegá esa URL en el campo de arriba
const SHEET_ID = '1V-hIDFcIloQhgNA9SQ3j9MIAqXCvScizinmR7I-nPrY';

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const sheet = SpreadsheetApp.openById(SHEET_ID)
                    .getSheetByName(data.hoja);
    
    if (!sheet) throw new Error('Hoja no encontrada: ' + data.hoja);
    
    // Si no hay encabezados, los agrega
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(data.headers);
    }
    
    sheet.appendRow(data.row);
    
    return ContentService
      .createTextOutput(JSON.stringify({ ok: true }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  return ContentService
    .createTextOutput(JSON.stringify({ ok: true, msg: 'Panel Luisito Creaciones activo' }))
    .setMimeType(ContentService.MimeType.JSON);
}
